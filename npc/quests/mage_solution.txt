pay_arche,122,100,0	script	아르바이트 직원#ab01	88,{
	mes "[마법 길드원]";
	mes "수용액이 필요한거야? 50 제니에 빈";
	mes "시험관 1 개 이상이 필요해.";
	next;
	if (select("수용액을 산다.:그만둔다.") == 1) {
		mes "[마법 길드원]";
		if (Zeny < 50) {
			mes "50 제니라니깐.";
			close;
		}
		if (countitem(1092) == 0) {
			mes "수용액을 어떻게 담아가려고 하는거야?";
			mes "빈 시험관을 가져와.";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
