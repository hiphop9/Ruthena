//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[존다 기획사 직원]";
	mes "어서오십시오. 존다 기획사는 언제나";
	mes "최고의 서비스를 하고 있습니다.";
	mes "어떤 서비스를 원하십니까?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[존다 기획사 직원]";
		mes "저장 되었습니다.";
		mes "이용해 주셔서 감사합니다.";
		mes getarg(1)+".";
/*		mes "Thank you for using the";
		mes "Cool Event Corp. service~";*/
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[존다 기획사 직원]";
		if(!callfunc("F_CanOpenStorage")){
			mes "죄송합니다. 고객님";
			mes "창고를 이용하시려면 초보자";
			mes "스킬레벨을 6 이상 올리시거나 새로운";
			mes "기본기능 스킬을 익히셔야 합니다.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "죄송합니다. 고객님";
			mes "창고를 이용하시려면 40제니가";
			mes "필요합니다.";
			close2;
			break;
		}
		mes "이용하실 창고를 선택 해 주세요.";
		next;
		switch(select("1번 창고:2번 창고")) {

		case 1:
			set .@sno,0;
			break;
		case 2:
			set .@sno,1;
			break;
		}
	
		callfunc("F_CheckKafCode");	//check your storage password, if set

		mes "창고를 열겠습니다.";
		mes "이용해 주셔서 감사합니다.";
		close2;
		set Zeny, Zeny - 40;
	
		if(.@sno <= 0)
			openstorage;
		else
			openstorage2 1,STOR_MODE_GET|STOR_MODE_PUT;
		
		break;
	case 3:
		mes "목적지를 선택하세요.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:취소") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "베인스") warp "veins",205,101;
				else if (.@destination$ == "라헬") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[존다 기획사 직원]";
				mes "죄송합니다. 고객님,";
				mes ""+ .@destination$ +"으(로) 공간이동 서비스를";
				mes "받기 위해서는 "+.@cost+" 제니가 필요합니다.";
			}
		}
		close2;
		break;
	case 4:
		mes "[존다 기획사 직원]";
		if(BaseClass != Job_Merchant){
			mes "죄송합니다.";
			mes "카트는 상인계열 직업군에 속한 분들만";
			mes "사용하실 수 있습니다.";
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "죄송합니다.";
			mes "카트를 사용하실려면 푸쉬카트";
			mes "스킬을 배우고 오셔야 합니다.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "이미 카트를 사용중입니다.";
			mes "다음에 다시 이용해주세요.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "카트를 대여 하는데 필요한 비용이";
		mes "800 제니가 필요합니다.";
		mes "카트를 대여하시겠습니까?";
		next;
		if(select("카트를 대여받는다:취소") == 1){
			if(Zeny < 800){
				mes "[존다 기획사 직원]";
				mes "죄송합니다.";
				mes "카트를 대여하기 위해서는";
				mes "800 제니가 필요합니다.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[존다 기획사 직원]","존다 기획사");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[존다 기획사 직원]";
		mes "저희 존다 기획사는 언제나 최고의";
		mes "서비스를 제공하고 있습니다. 혹시";
		mes "불편한점이 있으시면 언제든지 말씀해";
		mes "주시길 바랍니다.";
		mes "다음엔 보다 나은 서비스로";
		mes "찾아뵙겠습니다.";
		close2;
	}
	cutin "", 255;
	end;
}
