//================================================ ============
//= 메인 리파이너 함수
//================================================ ============
//= 자동 안전 정제/다중 정제를 허용하려면
//= 함수 호출에서 '1'에 대한 두 번째 인수입니다.
//= 이 기능을 활성화하는 경우 값을 편집해야 합니다.
//= .@safe를 통해 Refine.yml의 최대 안전 구체화도 가능합니다.
//================================================ ============
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "나는 대장장이입니다.";
	mes "모든 종류의 무기, 방어구, 장비를 제련할 수 있으니 맡겨주세요";
	mes "내가 무엇을 개선하고 싶은지 알아보세요.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "당신이 가지고 있는 어떤 아이템도 개선할 수 없을 것 같아요...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //사용자 정의 확인
		mes "["+ .@npc_name$ +"]";
		mes "당신은 옷을 입지 않았습니다";
		mes "그거 뭐든지";
		mes "수정할 수 있어요.";
		emotion ET_FRET;
		close;
	}
	//아이템이 정제 가능한지 확인합니다...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "난 할 수 없을 것 같아";
		mes "이 항목을 전혀 개선하지 마세요...";
		close;
	}
	//항목이 이미 +10인지 확인합니다.
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "이건 개선할 수가 없어요";
		mes "더 이상. 이건";
		mes "최선을 다해!";
		close;
	}
	.@refineitemid = getequipid(.@part); // 항목의 ID를 저장합니다.
	.@refinerycnt = getequiprefinerycnt(.@part); //정제소 개수 저장
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// 할 것:
				close;
		}

		// VIP 시스템이 활성화되면 VIP가 아닌 플레이어의 가격이 상당히 높아집니다.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// 할 것:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// 할 것:
				close;
		}

		// VIP 시스템이 활성화되면 VIP가 아닌 플레이어의 가격이 상당히 높아집니다.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: 이게 뭐였지? 아마도 그림자?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "이것을 개선하려면 필요해요";
		mes "one ^003366"+getitemname(.@material)+"^000000 and";
		mes "서비스 수수료 " + .@price + " Zeny.";
		mes "정말 계속하시겠습니까?";
		next;
		if(select("예:아니요") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "그래...";
			mes "그럴 필요 없어요";
			mes "서두르세요. 천천히 하세요.";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "안돼! 계속하면";
			mes "이것을 다듬으면 위험할 수 있어요";
			switch(.@material) {
			case 985:
				mes "파괴됩니다! 즉, ^FF0000이 장비^000000과 ^FF0000모든 카드^000000 또는 이 방어구에 추가된 특수 속성 ^FF0000^000000이 사라집니다.";
				break;
			default:
				mes "파괴되면 ^FF0000무기^000000, 무기에 있는 ^FF0000카드^000000을 잃게 됩니다,";
				mes "또는 추가된 특별한 속성.";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "더 명확하게 말할 수는 없습니다.";
			mes "무기가 파괴되면,";
			mes "돌려받을 수 없어요.";
			mes "정말 그럴 기회가 있어요";
			mes "^FF0000이 무기는 영원히 잃어버리세요^000000.";
			mes "아직도 개선하시겠습니까?";
			next;
			if(select("예:아니요") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "전적으로 동의합니다...";
				mes "나는 훌륭한 정제자이지만 때로는 실수를 할 때도 있습니다.";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {

function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "거기 여행자! 아이템 수리를 하고!";

	mes "아이템 수리는 나에게 맡기라구!";

	switch(select("사실 수리할 아이템이 있는데...:지금은 수리할게 없어요.")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "이런 이런. 자네는 정말 물건을";
			mes "잘다루는 모양이군. 손상된 아이템이";

			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "흠...";
		mes "어디보자...";
		mes "모든 아이템 중,";
		mes "" + .@checkitem + "가 손상되었습니다.";
		mes "수리하시겠습니까?";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "수리 비용은 " + .@repairprice + " Zeny입니다. 손상된 아이템을 모두 수리하려면 " + .@totalcost + " Zeny! 수리하시겠습니까?";
		next;
		switch(select("예:아니요")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "워워...";
				mes "수리비 청구서를 받기 전에 지갑을 확인해 보세요! 제니가 부족해서 수리를 할 수 없습니다.";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "좋아! 다 끝났어. 이제 조금 더 조심해. 아이템에도 생명이 있어.";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "으응? 뭔가 문제가 생겼어. 잠깐만... 수리할 아이템을 장착하고 다시 오너라.";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "뭐, 코가 벗겨지는 건 아니지만, 물건을 훼손한 채로 놔두는 건 좋지 않아요. 빨리 수리를 받아야 해요!";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "허허허... 수리할 아이템이 없다면";
		mes "나한테 볼일은 없는걸세.";
		close;
	}
}
