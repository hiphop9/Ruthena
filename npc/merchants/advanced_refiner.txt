payon,157,146,6	script	선비::Suhnbi#cash	85,{
	disable_items;
	mes "[선비]";
	mes "나는 대장장이입니다";
	mes "나는 모든 종류의 무기를 제련할 수 있어요,";
	mes "방어구와 장비를 알려주세요";
	mes "무엇을 개선하고 싶은지 알아보세요.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i])) {
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@equipped,1;
		}
		set .@menu$, .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "[선비]";
		mes "당신이 가지고 있는 어떤 아이템도 개선할 수 없을 것 같아요...";
		close;
	}
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) //custom check
		close;
	if (!getequipisenableref(.@part)) {
		mes "[선비]";
		mes "가서 다른 대장장이를 찾아보세요. 이건 개선할 수 없습니다.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[선비]";
		mes "흠... 이미 누군가가 완성한 부분인데 더 이상 작업할 수 없을 것 같아요.";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@price = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	// Make sure you have the necessary items and Zeny to refine your items
	// Determines chance of failure and verifies that you want to continue.
	callsub S_RefineValidate,.@itemtype,.@material,.@price,.@part,.@refineitemid,.@refinerycnt;

	mes "[선비]";
	mes "꽝! 꽝! 꽝! 꽝!";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[선비]";
		mes "자, 됐어요! 끝났어요.";
		mes "이렇게 잘 만든 건 오랜만이야. "+((.@itemtype == IT_WEAPON)?"weapon":"armor")+". 강해졌으니 기뻐하겠지!";
		close;
	}
	failedrefitem .@part;
	next;
	emotion (!rand(5))?ET_MONEY:ET_HUK;
	mes "[선비]";
	mes "우우우우우우우우우우우우우우우!!!";
	next;
	mes "[선비]";
	mes "...";
	mes ".....";
	mes ".......후후후후후~";
	mes "........당신의 선택이었고 나의 능력이었으며 후회는 없습니다.";
	close;

S_RefineValidate:
	.@itemtype = getarg(0);
	.@item_req = getarg(1);
	.@price = getarg(2);
	.@part = getarg(3);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch(.@equip_lv) {
				case 1:
					set .@price, .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch( .@equip_lv ){
				case 1:
					set .@price, .@price * 40;
					break;
				case 2:
					set .@price, .@price * 50;
					break;
				case 3:
					set .@price, .@price * 2;
					break;
				case 4:
					set .@price, .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		// TODO:
		close;
	}

	mes "[선비]";
	if (.@itemtype == IT_WEAPON)
		mes "레벨을 세분화하고 싶습니다. "+ .@equip_lv +" 무기?";
	mes "이를 개선하려면 다음이 필요합니다.^ff9999"+ getitemname(.@item_req) +"^000000 그리고"+ .@price +" 제니.";
	mes "계속하시겠습니까?";
	next;
	if(select("예 아니오") == 1) {
		if (getequippercentrefinery(.@part) < 100) {
			if (.@itemtype == IT_WEAPON) {
				mes "[선비]";
				mes "와!!";
				mes "이 무기는 아마";
				mes "세련된 것 같네요...";
				mes "여러번...";
				mes "깨질 수도 있어요";
				mes "다시 다듬으시네요.";
				next;
				mes "그리고 깨지면,";
				mes "더 이상 사용할 수 없어요!";
				mes "그 안에 있는 모든 카드와 속성 ^ff0000^000000이 손실됩니다!";
				mes "^ff0000게다가 장비가 망가질 수도 있어요!^000000";
				mes "정말 계속하시겠습니까?";
				next;
				if(select("예 아니오") == 2) {
					mes "[선비]";
					mes "좋아요.";
					mes "무리한 제련으로 무기가 망가지면 나도 기분이 나빠지니까.";
					close;
				}
			} else {
				mes "[선비]";
				mes "낄낄낄. 낄낄거림. 아, 용기가 있어 이걸 다듬으려는구나.";
				mes "이건 꽤 위험한 일인거 아시죠?";
				next;
				mes "방어장비가 파손되면 다시는 사용할 수 없습니다.";
				mes "당신의 카드와 수정사항도 ^ff0000완전히 사라집니다^000000.";
				//mes "Everything will disappear. As in... GONE!";
				mes "정말 계속하시겠습니까?";
				next;
				if(select("예 아니오") == 2) {
					mes "[선비]";
					mes "말도 안 되는 소리야. 넌 내 소중한 시간을 낭비하고 있어.";
					mes "길을 잃어라, 새끼야.";
					close;
				}
			}
		}
		if (countitem(.@item_req) > 0 && Zeny > .@price) {
			delitem .@item_req,1;
			set Zeny, Zeny - .@price;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@part, getarg(4)) ||
				callfunc("F_IsEquipRefineHack", .@part, getarg(5)) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
				mes "[홀링크]";
				emotion ET_FRET;
				mes "잠깐만요...";
				mes "내가 바보라고 생각해요?!";
				mes "내가 보지 않는 동안 아이템을 바꾸셨네요! 여기서 나가세요!";
				close;
			}

			return;
		}
		mes "[선비]";
		mes "이게 다예요?";
		mes "죄송하지만 재료가 없으면 아무것도 할 수 없어요. 게다가 작업에 대한 대가를 받을 자격도 있지 않나요?";
		close;
	}
	mes "[선비]";
	mes "네가 마음에 들지 않아도 어쩔 수가 없어...";
	close;
}
