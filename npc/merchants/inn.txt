//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "어서 오세요! 여관 "+ getarg(1) + "입니다.!";
	mes "무엇을 도와드릴까요?";
	next;
	switch(select("저장한다:쉬고간다 - 5000z:그만둔다")) {
	case 1:
		mes .@npc_name$;
		mes "저장하였습니다!";
		mes "확인 부탁드립니다!";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "여관 이용료는 5000z 입니다!";
			mes "확인 부탁드립니다!";
			close;
		}
		mes "5000z 확실히 받겠습니다!";
		mes "좋은 시간 되세요!";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
