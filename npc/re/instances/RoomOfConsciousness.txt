1@mir,103,80,3	script	펜릴#boss1a	4_F_FENRIR,2,2,{
	end;
OnEvent:
	'fenrir_name$ = instance_npcname( strnpcinfo(0) );
	'iris_name$ = instance_npcname( "아이리스#" + strnpcinfo(2) );
	enablenpc 'fenrir_name$;
	enablenpc 'iris_name$;
	end;

OnTouchNPC:
	if ('touch_mob == 0) {
		npctalk "펜릴: 잘하셨어요! 아이리스 부탁해요!", 'fenrir_name$;
		'touch_mob = 1;
		initnpctimer;
	}
	end;
OnTimer2000:
	npctalk "아이리스: 좋아 래볼게! 나 아이리스 아이린은 원한다!", 'iris_name$;
	specialeffect EF_BEGINSPELL,AREA, 'iris_name$;
	end;
OnTimer4000:
	npctalk "아이리스: 마비 부적! 에잇!!!", 'iris_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "으아아아악!!";
	else
		unittalk 'boss_id, "비쥬: 뭐야? 이건...";
	end;
OnTimer5000:
	npctalk "아이리스: 먹혔어!!", 'iris_name$;
	end;
OnTimer7000:
	npctalk "펜릴: 그럼 제 차례군요. 하아아아앗!", 'fenrir_name$;
	specialeffect EF_BEGINSPELL,AREA, 'fenrir_name$;
	specialeffect EF_TETRACASTING,AREA, 'fenrir_name$;
	progressbar_npc "000000",3;
	end;
OnTimer10000:
	npctalk "펜릴: 받아라!!", 'fenrir_name$;
	specialeffect EF_SUI_EXPLOSION,AREA, 'fenrir_name$;
	specialeffect EF_LORD,AREA, 'fenrir_name$;
	specialeffect EF_FLAMELAUNCHER,AREA, 'fenrir_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "아아아악!!";
	else
		unittalk 'boss_id, "비쥬: 윽... 생각보다 아픈 걸?";
	getunitdata 'boss_id, .@data;
	if (.@data[UMOB_HP] > 0) {
		.@damage = rand(700,1300) * 1000;
		if (.@damage >= .@data[UMOB_HP])
			.@mob_hp = 0;
		else
			.@mob_hp = .@data[UMOB_HP] - .@damage;
		setunitdata 'boss_id, UMOB_HP, .@mob_hp;
	}
	end;
OnTimer13000:
	npctalk "펜릴: 완벽하진 않았지만 큰 데미지를 준 것 같아요!", 'fenrir_name$;
	end;
OnTimer14500:
	npctalk "펜릴: 마법 소모가 심해서 잠시 쉬고 있을게요. 그 동안 녀석을 부탁해요!", 'fenrir_name$;
	end;
OnTimer16500:
	donpcevent instance_npcname("eq#mir3") + "::OnStart";
OnStop:
	stopnpctimer;
	disablenpc instance_npcname( strnpcinfo(0) );
	disablenpc instance_npcname( "아이리스#" + strnpcinfo(2) );
	'touch_mob = 0;
	'random_letter$ = "";
	end;
}
