niflheim,201,187,3	script	니플헤임 가이드#01nif	849,{
	mes "[니플헤임 가이드]";
	mes "^8B4513니플헤임^000000안내를...." ;
	mes "해...드리고.. 있습니다..";
	mes "무엇이든... 물어보세요..";
	if (F_Navi() == 0) {
		next;
		mes "[니플헤임 가이드]";
		mes "^B9062F[거점명]을 클릭^000000 시 ^B9062F내비게이션^000000 안내";
		mes "서비스...도 실시하고 있...습니다.";
	}
	while (1) {
		next;
		switch(select("[        도 구 점        ]:[          주 점          ]:[        무 기 점        ]:[      마 녀 의 성      ]:미니 맵 표시 지우기:그만둔다.")) {
		case 1:
			callsub L_Mark, F_Navi("도구점","niflheim,218,196","^0000FF");
			viewpoint 1,218,196,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("주점","niflheim,189,208","^006400");
			viewpoint 1,189,208,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("무기점","niflheim,218,170","^800080");
			viewpoint 1,218,170,2,0xDA70D6;
			break;
		case 4:
			callsub L_Mark, F_Navi("마녀의성","niflheim,255,195","^FF1493");
			viewpoint 1,255,195,3,0xFF1493;
			break;
		case 5:
			mes "[니플헤임 가이드]";
			mes "표시를.. 지워.. 드리겠습니다..";
			mes "다른 도움이... 필요하신가요..";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			break;
		case 6:
			mes "[니플헤임 가이드]";
			mes "......안녕...히..";
			close;
		}
	}
	end;
L_Mark:
	mes "[니플헤임 가이드]";
	mes getarg(0)+"의..";
	mes "위치를..미니.. 맵에";
	mes "표시해... 드리겠습니다.";
	mes "다른.. 안내가.. 필요하신가요?";
	return;
}
