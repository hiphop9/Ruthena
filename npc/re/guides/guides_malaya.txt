	mes "[포트 말라야 가이드]";

	mes "드리겠습니다.";
	next;
	while(1) {
		next;
		switch(select("[ 카프라 직원 ]:[ 여관 & 주점 ]:[   무기 상점  ]:[   도구 상점  ]:[     병  원     ]:[     공  원     ]:[     항  구     ]:미니 맵 표시 지우기:그만둔다.")) {

			mes "^006400카프라 직원^000000의";
			mes "위치는 미니맵의 ^0000FF상자모양 아이콘^000000을";
			mes "클릭하시면 안내 됩니다.";

			viewpoint 1,71,79,0,0x0A82FF;

			callsub L_Mark, F_Navi("여관&주점","malaya,193,215","^006400");
			viewpoint 1,193,215,2,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("무기상점","malaya,114,212","^008080");
			viewpoint 1,114,212,3,0x4F543C;
			break;
		case 4:
			callsub L_Mark, F_Navi("도구상점","malaya,297,167","^FF1493");
			viewpoint 1,297,167,4,0xFF1493;
			break;
		case 5:
			callsub L_Mark, F_Navi("병원","malaya,50,75","^8B4513");
			viewpoint 1,50,75,5,0x8B4513;
			break;
		case 6:
			callsub L_Mark, F_Navi("공원","malaya,280,371","^8B4513");
			viewpoint 1,280,371,6,0x8B4513;
			break;
		case 7:
			callsub L_Mark, F_Navi("항구","malaya,276,55","^8A2BE2");
			viewpoint 1,276,55,7,0x8A2BE2;
			break;
		case 8:
			mes "[포트 말라야 가이드]";
			mes "표시를 지워 드리겠습니다.";
			mes "다른 도움이 필요하세요?";

	}
	end;
