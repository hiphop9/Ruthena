				mes "어? 내가 당신을 아는가? 너무 낯익은 것 같아 소름끼친다. 쌍둥이는 없지?";

					next;

				}

				mes "글쎄요, 당신이 평생 싸움을 겪은 것 같아서 지금 당장 당신을 도둑으로 승급시키려고 합니다. 터프가이들에게 당신이 원하는 것을 주는 것이 좋을 것 같아요...";

				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[도둑 가이드]";
				mes "도둑이 되었으니 도둑으로 살아라. 자, 해보자! 다음~";
				close;
			}else{
				mes "[도둑 가이드]";
				if (Sex == SEX_MALE)
					mes "안녕 친구.";
				else
					mes "안녕 자기야~";
				close;
			}
		}else{
			mes "[도둑 가이드]";
			if (Sex == SEX_MALE)
				mes "안녕 친구.";
			else
				mes "안녕, 자기야.";
			mes "...야! 도둑이 되기엔 너무 착해 보이는데!! 자, 난 바쁘다. 다음!";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[도둑 길드맨]";
		mes "에에...부잣집에서 사랑하는 "+(Sex == SEX_MALE?"son":"daughter")+" 같은데 왜 이런 더러운 곳에 있는 걸까요?";
		next;
		mes "[도둑 길드맨]";
		mes "글쎄요, 저는 당신을 도둑으로 만드는 일을 맡는 것이 아닙니다. 단지 지원서를 받는 것뿐입니다. 알겠죠?";
		mes "도둑이 되고 싶다면 내 옆에 있는 눈썰미 좋은 놈에게 물어보세요.";
		close;
	}
	mes "[도둑 길드맨]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "야~ 무슨 일 있으면 언제든지 얘기해, 응?";
			close;
		}else{
			mes "도대체 무슨...?";
			if(Sex == SEX_MALE) {
				mes "어... 너 "+Jobname(Class)+"이구나, 멍청아!";
				next;
				mes "[도둑 길드맨]";
				mes "안녕하세요, 형제님.";
			}else{
				next;
				mes "[도둑 길드맨]";
				mes "안녕하세요 아가씨.";
			}
			mes "왜 왔어? 제자리로 돌아가~ 돌아가~~";
			close;
		}

		mes "[도둑 길드맨]";
		mes "신청 없이는 도둑이 될 수 없다는 거 아시죠...";
		mes "무슨 생각을 하고 계시나요...?";
		close;
	}
	mes "[도둑 길드맨]";
	mes "글쎄, 그렇게 자랑스러워?";
	mes "도둑이 되고 싶다고 자랑스럽게 말하잖아! 마을마다 다니면서 도둑이라는 광고를 하는 게 어때?";
	mes "'하하하! 가서 '나는 남의 물건을 훔치는 자랑스러운 도둑이 될 것이다'라고 적힌 현수막을 걸어라.";
	next;
	mes "[도둑 길드맨]";
	mes "그렇게 도둑이 되고 싶냐?";
	next;
	switch(select("예.:아니요.:당신은 어때요?")) {
	case 1:
		mes "[도둑 길드맨]";
		mes "아, 혹시...? 응.. 뭐... 난 너 같은 이상한 사람들을 보면서 배우며 살아가고 있어.";
		next;
		break;
	case 2:
		mes "[도둑 길드맨]";
		mes "그럼 넌 왜 여기 있는 거야? 그렇게 쉽게 도둑이 될 수 있을 것 같아?";
		next;
		break;
	case 3:
		mes "[도둑 길드맨]";
		mes "어..? 나? 나?";
		mes "글쎄요... 저는 도둑이 되는 게 딱 맞는 것 같아요... 특성상... 이런 어리석은 일에는 별로 개의치 ??않습니다.";
		next;
		break;
	}
	mes "[도둑 길드맨]";
	mes "그나저나 바깥세상에서는 절대 도둑이 되고 싶다고 말하지 마세요!!";
	next;
	mes "[도둑 길드맨]";
	mes "그럼 도둑이 되기 위해 지원하시겠습니까?";
	next;
	if(select("예, 그렇습니다.:아니요.") == 2) {
		mes "[도둑 길드맨]";
		mes "그럼... 하고 싶은대로 하세요~ 가세요~";
		close;
	}
	mes "[도둑 길드맨]";
	mes "알았어, 이름을 말해봐. 응? 뭐라고?";
	mes ""+strcharinfo(0)+". 그럼 "+strcharinfo(0)+" 맞죠...?";
	next;

		mes "[도둑 길드맨]";
		mes "도둑이 되려는 강한 의지가 보이네요......";
		mes "하지만 의지만으로는 실제 싸움에 나갈 수 없잖아요?";
		mes "그러니 가서 기본 스킬 레벨 9 이상을 달성하세요.";
		close;
	}
	mes "[도둑 길드맨]";
	mes "흠.... 탐정사무소 자료에서 흥미로운 범죄기록을 찾았네요. 그럼 당신은 좋은 도둑이 될 자질을 갖춘 것 같군요.";
	mes ""+strcharinfo(0)+". 이제 당신이 면접에 합격했음을 선언합니다!";
	next;

	mes "글쎄요, 저는 당신을 도둑으로 만드는 일을 맡는 것이 아닙니다. 단지 지원서를 받는 것뿐입니다. 알겠죠?";

	close;
}
