//반환: 주사위 값 [1..6] arg2 = 1인 경우에만
function	script	F_KO_Survival_dice	{
	mes "위치: " + getarg(0);
	mes "총 LP: " + job_kagero_lp;
	mes "주사위를 굴리려면 " + getarg(1,1) + " ^FD0260LP^000000.을 소모해야 합니다.";

	select("주사위를 굴려보세요.");
	if (job_kagero_lp < getarg(1,1)) {
		mes "^FD0260LP^000000이 부족하여 시작위치로 돌아갑니다.";
		callfunc "F_KO_Survival_warp",0;
		end;
	}
	set job_kagero_lp, job_kagero_lp - getarg(1,1);
	set .@dice, rand(1,6);
	emotion ET_OTL + .@dice, playerattached();
	mes "주사위는 " + .@dice + ".로 나왔습니다.";
	if (getarg(2,0))
		return .@dice;
	else {
		mes "앞으로 이동하려면 닫기를 누르세요. " + .@dice + " 블록.";
		callfunc "F_KO_Survival_warp", .@dice + getarg(0);
		end;
	}
}
