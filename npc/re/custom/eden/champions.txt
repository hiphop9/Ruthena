QuestComplete:
	mes "[챔피언 몬스터 사냥 의뢰 게시판]";

	mes " ";
	mes "보상: Exp "+getarg(4)+", Jexp "+getarg(5)+"";
	if($EdenQuestEvent == 1 && $EdenQuestPer > 1) {
		
		.QEExp = ((getarg(4) / 100)*$EdenQuestPer)+getarg(4);
		.QEJExp = ((getarg(5) / 100)*$EdenQuestPer)+getarg(5);

		mes "^7882ff보상("+$EdenQuestPer+"%): Exp "+.QEExp+", Jexp "+.QEJExp+"^000000";
	}
	else {
		.QEExp = 0;
		.QEJExp = 0;
	}
	if(getarg(6) > 0)
		mes ""+getitemname(getarg(6))+" "+getarg(7)+"개";
	
	erasequest getarg(0);
	setquest getarg(1);
	getexp getarg(4),getarg(5);
	getexp .QEExp,.QEJExp;
	if(getarg(6) > 0)
		getitem getarg(6),getarg(7);
	next;
	return;
