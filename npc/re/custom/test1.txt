prontera,151,191,5	script	전직도우미	897,{

    mes "Welcome to the Job Changer.";
    next;

    if (Class == 0) goto FirstJob;  // 1차 전직
    if (Class >= 1 && Class <= 6) goto SecondJob;  // 2차 전직
    if (Class >= 4008 && Class <= 4022) goto ThirdJob;  // 3차 전직
    if (Class >= 4054 && Class <= 4108) goto FourthJob;  // 4차 전직
    if (Class == 22 || Class == 23) goto BabyJob;  // 베이비 전직
    close;

FirstJob:
    mes "[Class Master]";
    mes "You are eligible to change to a first job.";
    next;
    menu "Swordman", SWORDMAN, "Magician", MAGICIAN, "Archer", ARCHER, "Acolyte", ACOLYTE, "Merchant", MERCHANT, "Thief", THIEF;
    
    SWORDMAN:
        jobchange 1;
        mes "You are now a Swordman!";
        close;

    MAGICIAN:
        jobchange 2;
        mes "You are now a Magician!";
        close;

    ARCHER:
        jobchange 3;
        mes "You are now an Archer!";
        close;

    ACOLYTE:
        jobchange 4;
        mes "You are now an Acolyte!";
        close;

    MERCHANT:
        jobchange 5;
        mes "You are now a Merchant!";
        close;

    THIEF:
        jobchange 6;
        mes "You are now a Thief!";
        close;

    if (JobLevel < 40) {
        mes "[Class Master]";
        mes "You need at least Job Level 40 to advance to the second job.";
        close;
    }
    
    mes "[Class Master]";
    mes "You are eligible to change to a second job.";
    next;
    menu "Knight", KNIGHT, "Wizard", WIZARD, "Hunter", HUNTER, "Priest", PRIEST, "Blacksmith", BLACKSMITH, "Assassin", ASSASSIN;

    KNIGHT:
        jobchange 7;
        mes "You are now a Knight!";
        close;

    WIZARD:
        jobchange 8;
        mes "You are now a Wizard!";
        close;

    HUNTER:
        jobchange 9;
        mes "You are now a Hunter!";
        close;

    PRIEST:
        jobchange 10;
        mes "You are now a Priest!";
        close;

    BLACKSMITH:
        jobchange 11;
        mes "You are now a Blacksmith!";
        close;

    ASSASSIN:
        jobchange 12;
        mes "You are now an Assassin!";
        close;

ThirdJob:
    if (BaseLevel < 99 || JobLevel < 50) {
        mes "[Class Master]";
        mes "You need at least Base Level 99 and Job Level 50 to advance to the third job.";
        close;
    }

    mes "[Class Master]";
    mes "You are eligible to change to a third job.";
    next;
    menu "Rune Knight", RUNEKNIGHT, "Warlock", WARLOCK, "Ranger", RANGER, "Archbishop", ARCHBISHOP, "Mechanic", MECHANIC, "Guillotine Cross", GUILLOTINECROSS;

    RUNEKNIGHT:

        mes "You are now a Rune Knight!";
        close;

    WARLOCK:
        jobchange 4055;
        mes "You are now a Warlock!";
        close;

    RANGER:
        jobchange 4056;
        mes "You are now a Ranger!";
        close;

    ARCHBISHOP:
        jobchange 4057;
        mes "You are now an Archbishop!";
        close;

    MECHANIC:
        jobchange 4058;
        mes "You are now a Mechanic!";
        close;

    GUILLOTINECROSS:
        jobchange 4060;
        mes "You are now a Guillotine Cross!";
        close;

FourthJob:
    if (BaseLevel < 200 || JobLevel < 70) {
        mes "[Class Master]";
        mes "You need at least Base Level 200 and Job Level 70 to advance to the fourth job.";
        close;
    }

    mes "[Class Master]";
    mes "You are eligible to change to a fourth job.";
    next;
    menu "Dragon Knight", DRAGONKNIGHT, "Archmage", ARCHMAGE, "Windhawk", WINDHAWK, "Cardinal", CARDINAL, "Biolo", BIOLO, "Shadow Cross", SHADOWCROSS;

    ARCHMAGE:

        close;

        jobchange 4089;
        mes "You are now a Windhawk!";
        close;

    CARDINAL:
        jobchange 4090;
        mes "You are now a Cardinal!";
        close;

    BIOLO:
        jobchange 4091;
        mes "You are now a Biolo!";
        close;

    SHADOWCROSS:
        jobchange 4093;
        mes "You are now a Shadow Cross!";
        close;

BabyJob:
    if (BaseLevel < 10) {
        mes "[Class Master]";
        mes "You need at least Base Level 10 to become a baby class.";
        close;
    }
    
    mes "[Class Master]";
    mes "You are eligible to change to a baby job.";
