//=== 함수: F_EpDesc ========================================
function	script	F_EpDesc	{
	.@ep = getarg(0);
	mes "[밝히리]";
	mes "^4d4dff에피소드 " + .@ep + "^000000?";
	switch (.@ep) {
		case 13: mes "이계와 디카스테스 관련된 이야기구나."; break;
		case 14: mes "비프로스트와 에클라쥬의 이야기구나."; break;
		case 15: mes "판타스마고리카와 발굴단 이야기구나."; break;
		case 16: mes "테라 글로리아를 둘러싼 이야기구나."; break;
		case 17: mes "일루시온과 바르문트의 이야기구나."; break;
		case 18: mes "회색늑대 마을과 관련된 이야기구나."; break;
		case 19: mes "이스가르드의 얼음성과 관련된 이야기구나."; break;
		case 20: mes "죽지 않는 자와 관련된 이야기구나."; break;
	}
	mes "잠깐 살펴볼게.";
	next;
	return;
}

//=== 함수: F_CheckQuestState ===============================
function	script	F_CheckQuestState	{
	.@id = getarg(0);
	.@state = checkquest(.@id, PLAYTIME);
	if (.@state == 2) return "^4d4dff완료^000000";
	return "미완료";
}

//=== 함수: F_EpClearSkip_SetVars ===========================
function	script	F_EpClearSkip_SetVars	{
	.@ep = getarg(0);
	if (.@ep == 15 && terra_gloria_main < 25) terra_gloria_main = 25;
	if (.@ep == 16 && terra_gloria_main < 26) terra_gloria_main = 26;
	return;
}
