icas_in,169,70,3	script	수납장#ep19	4_POINT_BLUE,{
	.@total_storage = 1;	// by default rAthena just open the basic storage (but 3 on kRO)

	cutin "ep19_iwin02.png",2;
	mes "[호요요]";
	mes "수납장 쓰려고효?";
	mes "효효효~ 대륙에서 쓰던 동전을 넣어봐효.";
	mes "호오... 500zeny 정도 넣어보겠어효?";
	next;
	cutin "",255;
	for ( .@i = 0; .@i < .@total_storage; .@i++ )
		.@menu$ += (.@i+1) + "번 창고" + ":";
	.@menu$ += "그만 둔다.";
	.@s = select( .@menu$ ) - 2;

	if (.@s == (.@total_storage - 1)) {
		mes "다음에 사용하기로 했다.";
		close;
	}

	if (Zeny < 500) {
		mes "수납장을 사용하려면 500zeny를 투입해야 하는 것 같다.";
		close3;
	}
	mes "동전을 투입하자 덜컥 소리와 함께 수납장이 열렸다.";
	close2;
	Zeny -= 500;
	if (.@s == -1)	// basic storage
		openstorage;
	else	// "premium" storage starting from index 0
		openstorage2 .@s, STOR_MODE_GET|STOR_MODE_PUT;	// note: opened before close button
	end;
}
