function	script	F_ep20_checkTransform	{
	if (getstatus(SC_MONSTER_TRANSFORM,1) != 21530) {
		if (getarg(0,0) == 1) {
			npctalk "인간??! 그때 그 인간이다!!", "", bc_self;
			setpcblock PCBLOCK_NPC, true;
			sleep2 1000;
			setpcblock PCBLOCK_NPC, false;
		}
		else {
			mes "[최상급 르간]";
			mes "인간이다!!";
			close2;
		}
		warp "jor_nest",38,263;
		end;
	}
	return;
}
