	mes "아카데미가 이곳 Izlude로 보내졌습니다";
	mes "스킬 재설정 서비스를 제공하기 위해";

	next;
	mes "[최면술사]";
	mes "무제한으로 제공하지만";
	mes "지금은 능력치/스킬을 재설정했습니다.";
	mes "반드시 있어야 하는 두 가지 조건";
	mes "완성되었습니다. 먼저, 성취되어야 합니다";
	mes "^FF0000베이스 레벨 50^000000보다 낮습니다.";
	next;
	mes "[최면술사]";
	mes "둘째, 당신은 a가 되어야 합니다";
	mes "^FF0000검사, 복사, 마법사,";
	mes "도둑, 궁수, 상인,";
	mes "태권 소년, 또는 태권 소녀,";
	mes "총잡이와 닌자^000000";
	mes "자격을 갖춘 직업 캐릭터입니다.";
	mes "이제 질문이 있으신가요?";
	next;
	switch(select("^FF0000스텟/스킬 재설정?:스텟 재설정:스킬 재설정^000000:취소")) {
	case 1:
		mes "[최면술사]";
		mes "스텟/스킬 재설정은 모험가를 허용합니다";
		mes "스킬을 재분배하기 위해";

		mes "현재 기술로.";

		mes "[최면술사]";
		mes "아, 그것도 중요해요";
		mes "^FF0000푸쉬카트^000000을 제거하려면";

		mes "일하지 않으면 역효과를 낳을 것이다...";

		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "[최면술사]";
		mes "정말인가요?";
		mes "계속하고 싶어요";

		if(select("Yes:Cancel") == 1)
			break;

		mes "[최면술사]";
		mes "고마워요, 좋아요";
		mes "당신의 모험에 행운을 빕니다.";
		mes "안전하게 여행하세요~";
		close;

		if (checkcart()) {
			mes "[최면술사]";
			mes "아! 삭제해주세요";
			mes "계속하기 전에 카트를 끌고 가세요";
			mes ""+.@str$+" 재설정 서비스를 사용합니다.";
			mes "협조해주셔서 감사합니다~";
			close;

			mes "[최면술사]";
			mes "나를 위해 여기 오셨다면"+.@str$;
			mes "서비스를 초기화해주세요";
			mes "당신은 할 수 없다는 것을 기억하세요";
			mes "다음이 될 때까지 "+.@str$+"을 재설정하세요";

			mes "당신의 인벤토리가 0^000000으로 줄었습니다.";
			next;
			mes "[최면술사]";
			mes "당신의 자리를 마련하는 게 어때요?";
			mes "카프라 저장소에 있는 것들";
			mes "지금은요? 그렇게 하면 할 수 있어요";
			mes "모든 물건을 안전하게 보관하세요.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[최면술사]";
			mes "죄송합니다만 캐릭터입니다";
			mes "기본 레벨이 더 높음";
			mes "50명 이상이면 자격이 없습니다";

			close;
		}
		mes "[최면술사]";
		mes "이용해주셔서 감사합니다";
		mes "나의 "+.@str$+" 재배포";
		mes "서비스. 아, 최고예요";
		mes "당신에게 행운이 있기를";
		mes "여행, 모험가.";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[최면술사]";
		mes "죄송하지만 당신";
		mes "직업이 자격을 갖추지 못했습니다";
		mes ""+.@str$+" 재설정 서비스용";
		mes "내가 제공하는 것. 나는 오직 할 수만 있다";
		mes "다음 "+.@str$+" 으로 재설정 권장합니다.";
		mes "작업 팔로우 중...";
		next;
		mes "[최면술사]";
		mes "^FF0000소드맨, 수행사제,";
		mes "마법사, 도둑, 궁수,";
		mes "상인, 태권";
		mes "소년, 태권소녀,";
		mes "건슬린더와 닌자^000000.";
		close;
	}
}
