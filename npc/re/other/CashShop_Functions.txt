function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[어디선가 들려오는 목소리]";
		mes "프리미엄 콘텐츠는 일정량의 프리미엄 리셋 돌을 보유한 사람만 이용할 수 있습니다.";
		next;
		if (countitem(6320) < 1) {
			mes "[어디선가 들려오는 목소리]";
			mes "3등급으로 변경하시려면 ^0000CC프리미엄 리셋 돌^000000 1개를 가져오거나 프리미엄 계정을 가지고 있어야 합니다.";
			//mes "^0000CC프리미엄 리셋 돌^000000 매하려면 캐시 샵을 방문하십시오..";
			close;
		} else {
			mes "[어디선가 들려오는 목소리]";
			mes "이 재설정 ^0000CC프리미엄 리셋 돌^000000 1개 퀘스트에 참여하시겠습니까??";
			next;
			switch(select("예:아니오")) {
			case 1:
				mes "[어디선가 들려오는 목소리]";
				mes "모험가,행운을 빌어요.";
				delitem 6320,1; //프리미엄 리셋 돌
				next;
				break;
			case 2:
				mes "[어디선가 들려오는 목소리]";
				mes "아주 좋아.";
				close;
			}
		}
	}
	return;
}
