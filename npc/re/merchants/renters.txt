		mes "환영합니다. 용을 대여하시겠습니까?";
		next;
		switch(select("예:아니요")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[용 키우는 사람]";
				mes "먼저 용을 타는 법을 배우셔야 합니다.";
				close;
			}
			else if (checkdragon()) {
				mes "[용 키우는 사람]";
				mes "이미 용을 소유하고 계십니다.";
				close;
			}
			else if (ismounting()) {
				mes "[용 키우는 사람]";
				mes "캐시 마운트를 해제해주세요.";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[용 키우는 사람]";
			mes "알겠습니다. 즐거운 시간 되세요.";
			close;
		}
	}
	mes "여기서 무슨 일을 하시나요?";
	mes "용을 대여할 수 있는 것은 룬 나이트뿐입니다.";
	close;
}

-	script	::mgm	-1,{
	mes "[마도 기어 제작자]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "환영합니다, 메카닉.";
		mes "푸시카트를 대여하시거나";
		mes "마도 기어를 타시겠습니까?";
		next;
		switch(select("푸시카트 대여:마도 기어 탑승:비상 마도 기어 구매:냉각 장치 업그레이드:취소")) {
		case 1:
			if (checkcart()) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 이미 푸시카트를 가지고 있습니다.";
				close;
			}
			setcart;
			mes "[마도 기어 제작자]";
			mes "여기 있습니다!";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 이미 마도 기어를 타고 있습니다.";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[마도 기어 제작자]";
				mes "먼저 마도 기어 라이선스 획득 스킬을 배우십시오.";
				close;
			}
			else if(ismounting()) {
				mes "[마도 기어 제작자]";
				mes "캐시 마운트를 제거해주세요.";
				close;
			}
			setmadogear;
			mes "[마도 기어 제작자]";
			mes "즐겁게 사용하세요, 언제든 다시 오세요!";
			close;
		case 3:
			mes "[마도 기어 제작자]";
			mes "비상 마도 기어는 긴급 상황에 매우 유용하며 1,000,000 제니에 판매됩니다.";
			next;
			if (select("구매:취소") == 2) {
				mes "[마도 기어 제작자]";
				mes "알겠습니다. 마음이 바뀌면 언제든지 물어봐주세요.";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 이미 비상 마도 기어를 가지고 있습니다.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 비상 마도 기어를 구매하기에는 충분한 제니가 없습니다.";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[마도 기어 제작자]";
			mes "여기 있습니다!";
			close;
		case 4:
			mes "[마도 기어 제작자]";
			mes "어떤 장치를 업그레이드하시겠습니까?";
			next;
			if (select("냉각 장치:고급 냉각 장치") == 1) {
				mes "[마도 기어 제작자]";
				mes "냉각 장치를 고급 냉각 장치로 업그레이드하려면 냉각 장치 1개와 2,000,000 제니가 필요합니다.";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[마도 기어 제작자]";
				mes "고급 냉각 장치를 스페셜 쿨러로 업그레이드하려면 고급 냉각 장치 1개와 4,000,000 제니가 필요합니다.";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("업그레이드:취소") == 2) {
				mes "[마도 기어 제작자]";
				mes "알겠습니다. 마음이 바뀌면 언제든지 물어봐주세요.";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 " + getitemname(.@itemid) + "을(를) 가지고 있지 않습니다.";
				close;
			}
			if (Zeny < .@cost) {
				mes "[마도 기어 제작자]";
				mes "죄송하지만 장치를 업그레이드하기에 충분한 제니가 없습니다.";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[마도 기어 제작자]";
			mes "여기 있습니다! 여러분의 " + getitemname(.@itemid) + "입니다.";
			close;
		case 5:
			close;
		}
	}
	mes "무엇을 도와드릴까요?";
	mes "마도 기어는 메카닉에게만 사용 가능합니다.";
	close;
}
