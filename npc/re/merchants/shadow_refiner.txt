//===== rAthena Script =======================================
//= Shadow Blacksmith
//===== Description: =========================================
//= [Official Conversion]
//= Shadow equipments refining NPC.
//===== Changelog: ===========================================
//= 1.0 First version [Aleos]
//= 1.1 Removed re-roll behavior and fetch materials from db
//=     [Secret]
//= 1.2 Added Refine UI [Atemo, Lemongrass]
//= 1.3 Moved to Cash Mall [Lemongrass]
//============================================================
-	script	::ShadowBlacksmith	-1,{
	if( getbattleflag( "feature.refineui" ) ){
		mes "[쉐도우 대장장이]";
		mes "이제 바스타는 섀도우 아이템의 세련을 담당합니다.";
		mes "섀도우 아이템을 다듬고 싶다면 방문해 보세요.";
		close;
	}

	.@zeny_cost = 200000; // Zeny cost is 200,000 according to official script [Secret]
	disable_items;
	mes "[쉐도우 대장장이]";
	mes "그림자 아이템을 개선하시겠습니까? 독을 선택하세요!";
	next;
	setarray .@indices[1], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;
	.@indlen = getarraysize(.@indices) - 1;
	for(.@i = 1; .@i <= .@indlen; .@i++)
		.@menu$ = .@menu$ + (getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Not equipped]") +":";
	.@menu$ = .@menu$ + "Refine info";
	.@choice = select(.@menu$);
	if (.@choice == .@indlen + 1) { // Refine info
		mes "[쉐도우 대장장이]";
		mes "그림자 아이템을 정제하면 일반 아이템과 마찬가지로 추가 보너스를 얻을 수 있습니다.";
		next;
		mes "[쉐도우 대장장이]";
		mes "무기 : 제련 1당 ATK, MATK 1씩 증가합니다.";
		mes "기타: 제련 성공 1당 HP 10 증가.";
		next;
		mes "[쉐도우 대장장이]";
		mes "모든 유형의 오리데오콘 및 에르늄을 사용하여 그림자 아이템을 다듬을 수 있습니다. 각 시도에는 20,000제니의 비용이 듭니다.";
		next;
		mes "[쉐도우 대장장이]";
		mes "HD 광석은 제련 레벨 7 이상인 장비에 사용할 수 있으며, 나와 대화를 계속하는 한 파손을 방지할 수 있습니다.";
		close;
	}

	.@part = .@indices[.@choice];

	if (!getequipisequiped(.@part)) {
		mes "[쉐도우 대장장이]";
		mes "여기에는 아무것도 없습니다!";
		close;
	}
	.@equip_id = getequipid(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	while(1) {
		.@equip_refine = getequiprefinerycnt(.@part);
		mes "[쉐도우 대장장이]";
		mes "각 개선 시도에 대한 수수료로 " + callfunc("F_InsertComma", .@zeny_cost) + " zeny가 필요합니다.";
		mes "광석을 선택하고 정제를 시작하세요.";
		next;

		.@material[0] = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
		.@material[1] = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
		.@material[2] = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);
		.@is_enriched = false;

		if (countitem(.@material[0]))
			.@mate$[0] = getitemname(.@material[0]);
		else
			.@mate$[0] = "^8C8C8C"+ getitemname(.@material[0]) +"^000000";
		if (countitem(.@material[1]))
			.@mate$[1] = getitemname(.@material[1]);
		else
			.@mate$[1] = "^8C8C8C"+ getitemname(.@material[1]) +"^000000";
		if (.@equip_refine > 6 && countitem(.@material[2]))
			.@mate$[2] = getitemname(.@material[2]);
		else
			.@mate$[2] = "^8C8C8C"+ getitemname(.@material[2]) +"^000000";

		switch( select( .@mate$[0], .@mate$[1], .@mate$[2], "Cancel" ) ) {
		case 1:// NORMAL
			.@choose = .@material[0];
			break;
		case 2:// ENRICHED
			.@is_enriched = true;
			.@choose = .@material[1];
			break;
		case 3:// HD
			if (.@equip_refine < 7) {
				mes "[쉐도우 대장장이]";
				mes "HD 광석은 7제련 이상 아이템에만 사용할 수 있습니다.";
				close;
			}
			.@hoihoi = 1;
			.@choose = .@material[2];
			break;
		case 4:
			mes "[쉐도우 대장장이]";
			mes "정제를 취소하셨습니다..";
			close;
		}
		if (!countitem(.@choose)) {
			mes "[쉐도우 대장장이]";
			mes "당신은 충분하지 않습니다 "+ getitemname(.@choose) +".";
			close;
		}
		if (Zeny < .@zeny_cost) {
			mes "[쉐도우 대장장이]";
			mes "제니가 부족합니다.";
			close;
		}
		if (.@equip_refine > 9) {
			mes "[쉐도우 대장장이]";
			mes "그림자 장비는 최대 10개까지 제련할 수 있습니다...";
			close;
		}
		if (!getequipisenableref(.@part)) {
			mes "[쉐도우 대장장이]";
			mes "이 아이템은 재련이 불가능합니다.";
			close;
		}
		if (getequippercentrefinery(.@part) < 100) {
			mes "[쉐도우 대장장이]";
			mes "그림자 장비의 안전 제련도는 4입니다.";
			if (!.@hoihoi)
				mes "섀도우 장비는 이후 개선 시 파괴될 수 있습니다. 시작하시겠습니까?";
			else
				mes "다음 정제가 실패하면 정제 수준이 저하됩니다. 개선하시겠습니까?";
			next;
			if (select("Proceed","Cancel") == 2) {
				mes "[쉐도우 대장장이]";
				mes "흥, 그럴 줄 알았어!";
				close;
			}
		}
		mes "[쉐도우 대장장이]";
		mes "그럼.. 아무것도 안 되네요!";
		next;
		delitem .@choose,1;
		Zeny -= .@zeny_cost;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
			close;

		if (getequippercentrefinery(.@part, .@is_enriched) > rand(100)) {
			successrefitem .@part;
			mes "[쉐도우 대장장이]";
			mes "효과가 있었어요! 성공했어요!";
			next;
		} else {
			if (.@hoihoi)
				downrefitem .@part;
			else
				failedrefitem .@part;
			mes "[쉐도우 대장장이]";
			mes "Oh Odin No!";
			close;
		}
	}
}
