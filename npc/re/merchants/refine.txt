//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "나는 최고의 Armsmith입니다!";
	mes "저는 평범하고 지루한 아이템은 개조하지 않아요.";
	mes "10레벨 이상의 아이템만 제련하고 있습니다.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "어차피, 아이템 레벨이 10 이상이면 서비스를 이용하실 수 있습니다.";
	mes "무엇을 다듬고 싶나요?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "당신이 가지고 있는 어떤 아이템도 개선할 수 없을 것 같아요...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "당신은 옷을 입지 않았습니다";
		mes "그거 뭐든지";
		mes "수정할 수 있어요.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "난 할 수 없을 것 같아";
		mes "이 항목을 전혀 개선하지 마세요...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "10레벨 이하의 아이템은 작업하지 않는다고 했는데요.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "이건 개선할 수가 없어요";
		mes "더 이상. 이건";
		mes "최선을 다해!";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "흠 " + .@article$ + " " + .@type$ + ", 그 확인은?";
	mes "이것을 다듬고 싶다면 " + .@type$ + ",";
	mes "1^003366이 필요합니다." + getitemname(.@material) + "^000000 그리고 " + callfunc("F_InsertComma",.@price) + " zeny.";
	mes "너 정말 계속하고 싶니?";
	next;
	if (select("Yes:No") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "흠... 괜찮다면... 신경쓰지 마세요...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "이것 "+.@type$+" 이미 여러번 다듬어져.";
			mes "다시 시도하면 파괴될 수 있습니다.";
			mes "확실히 깨지지는 않겠지만 가능성은 조금 있습니다.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "당신은 ^FF0000업그레이드 레벨을 낮출 수 있습니다^000000"+.@type$+",";
			mes "또는 깨지면 ^FF0000모든 카드^000000 또는 추가된 특수 속성을 잃게 됩니다.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "아직도 제가 다듬기를 원하시나요?";
			mes "충분히 경고한 것 같습니다.";
			next;
			if(select("예 아니오.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "글쎄, 도전에는 한 가지 방법만 있는 게 아니지...";
				mes "위험은 없어요... 그게 현명할 수도 있어요.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "흠. 돈이 부족한 것 같던데."+getitemname(.@material)+".";
			mes "갖고 나면 다시 오세요.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "잠깐만요...";
			mes "내가 바보라고 생각해요?!";
			mes "내가 보지 않는 동안 아이템을 바꾸셨네요! 여기서 나가세요!";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "꽝! 꽝! 꽝! 꽝! 꽝!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "좋아요! 성공했어요!!!";
			mes "나는 최고의 대장장이이다.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "그 소리! 그 소리! 그 소리! 그 소리!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "아아아!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "맙소사!";
				mes "업그레이드 레벨이 떨어졌어요...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "꽝! 꽝! 꽝! 꽝!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "흠!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "어머! 다듬는데 실패했어요...";
				mes "그런 뜻이 아니었어요!";
			}
			mes "내가 최고의 제련사임에도 불구하고 실수를 할 수도 있었습니다.";
			mes "그냥 그럴 의도는 아니었어요.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "다음에는 더 잘할께요! 괜찮아요!";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "이를 안전 한도 또는 원하는 횟수로 세분화할 수 있습니다. 그것은 당신의 선택입니다.";
		next;
		.@menu2 = select("안전한 한도까지 부탁드립니다.","몇번인지는 제가 정하겠습니다.","마음이 바뀌었어요...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "귀하의 아이템을 몇 번이나 수정하시겠습니까?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "이 아이템은 그렇게 여러 번 정제할 수 없습니다.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "이것은 장비를 개선하려고 노력할 것입니다 " + .@refinecheck + "안전 한도를 초과한 횟수입니다. 장비가 파손될 수 있으니...괜찮나요?";
			next;
			if(select("예...","아니요...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "그렇게 말했잖아요...그렇게 하세요.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "그렇게 말했잖아요...그렇게 하세요.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "그러면 비용이 들게 됩니다" + .@refinecnt + " " + getitemname(.@material) + " 그리고" + .@fullprice + " 제니. 그 확인은?";
	next;
	if(select("Yes:No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "그렇게 말했잖아요... 그렇게 하세요.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "흠. 돈이 부족한 것 같거나 "+getitemname(.@material)+".";
		mes "가지고 계시면 다시 오세요.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "여기 보세요... 아이템이 하나도 없어요...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "삑삑... 아니, 그런데 내가 이렇게 멍청할 수 있을 거라고 상상이나 했어?!";
			mes "바꾸셨군요...";
			mes "내가 망치로 널 기절시키기 전에 나가!!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "꽝! 꽝! 꽝! 꽝! 꽝!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "꽝! 꽝! 꽝! 꽝! 꽝!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "아아아!!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "맙소사!";
				mes "업그레이드 레벨이 떨어졌어요...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "꽝! 꽝! 꽝! 꽝!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "흠!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "어머! 다듬는데 실패했어요...";
				mes "그런 뜻이 아니었어요!";
			}
			mes "내가 최고의 Armsmith임에도 불구하고 실수를 할 수도 있었습니다.";
			mes "그냥 그럴 의도는 아니었어요.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "다음에는 더 잘할께요! 걱정하지 마세요!";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "좋아요! 성공했어요!!!";
	mes "나는 최고의 대장장이이다.";
	close;
}

	mes "[쨍그랑]";

			break;

			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "액세서리는 어디에 있나요?";
			break;

		case EQI_HEAD_LOW:
			mes "머리 부분을 말씀하시는 건가요?";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[쨍그랑]";
		mes "이건 정제할 수 없어!!";
		close;

	close;
