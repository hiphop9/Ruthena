-	script	GuildWarehouse	-1,{
	mes "[창고 관리자]";
	mes "안녕하세요~ 저희는 창고 전문,";
	mes "^0000FF스토리.G^000000입니다.";
	mes "길드멤버들끼리 공용으로 창고를 사용할 수 없을까? 하는 작은 의문에서 출발한 단체죠.";
	next;
	mes "[창고 관리자]";
	mes "1,000제니로 창고를 열거나 길드원의";
	mes "창고 이용 내역을 살펴보기를";
	mes "지원합니다.";
	mes "어떤 것을 도와드릴까요?";
	next;
	if( select( "길드 창고 열기:창고 이용 내역 보기" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[창고 관리자]";
			mes "길드창고는 길드에 가입한 길드원에게만 열어드립니다.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[창고 관리자]";
			mes "죄송하지만, 길드 창고를 사용 할 수 없거나, 길드 창고 사용 허가를 받지 못하신 것 같아요.";
			mes "길드 스킬과 창고 사용 권한을 다시 확인하시고 와주세요.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[창고 관리자]";
			mes "길드 창고 서비스를 사용하기 위한 비용이 충분하지 않아요.";
			mes "비용은 1,000제니 랍니다.";
			close;
		}

		mes "[창고 관리자]";
		mes "길드 창고를 바로 열어드릴께요!";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[창고 관리자]";
			mes "지금 다른 멤버가 길드 창고를 사용하고 있어요.";
			mes "조금 있다가 다시 시도해주실레요?";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[창고 관리자]";
			mes "길드창고는 길드 멤버만 사용가능 해요.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[창고 관리자]";
			mes "죄송하지만, 길드 창고를 사용 할 수 없거나, 길드 창고 사용 허가를 받지 못하신 것 같아요.";
			mes "길드 스킬과 창고 사용 권한을 다시 확인하시고 와주세요.";
			close;
		}

		mes "[창고 관리자]";
		mes "길드창고 사용 기록을 보여드릴께요. 사용 기록은 최대 3개월까지만 기록이 된답니다.";
		mes "그럼 좋은 하루 되세요!";
		close2;
		guildopenstorage_log();
		end;
	}
}
