bra_in01,27,24,3	script	호텔 지배인#bra1	478,{
	mes "[호텔 지배인]";
	mes "아름다운 브라질리스 호텔에 오신걸 환영합니다!";
	next;
	switch(select("저장:호텔 투숙(5,000제니)")) {
	case 1:
		mes "[호텔 지배인]";
		mes "저장 포인트를 브라질리스 호텔로 설정하시겠습니까?";
		next;
		switch(select("괜찮아요.:네.")) {
		case 1:
			mes "[호텔 지배인]";
			mes "알겠습니다. 즐거운 시간 되십시오.";
			close;
		case 2:
			mes "[호텔 지배인]";
			mes "리스폰 포인트가 브라질리스 호텔로 저장되었습니다. 이곳에 머무는동안 즐거운 시간 되시기 바랍니다.";
			savepoint "bra_in01",144,69,1,1;
			close;
		}
	case 2:
		if (Zeny > 4999) {
			mes "[호텔 지배인]";
			mes "좋은 방으로 안내하겠습니다.";
			close2;
			set Zeny, Zeny-5000;
			percentheal 100,100;
			warp "bra_in01",144,69;
			end;
		}
		else {
			mes "[호텔 지배인]";
			mes "죄송하지만, 숙박비는 1회에 5,000제니 랍니다.";
			close;
		}
	}
}
