//= [Official Conversion]
//= Refiner that uses Enriched ores to increase upgrade success.
//===== Changelog: ===========================================

//= 1.1 Removed re-roll behavior. [Secret]
//= 1.2 Added db-based material ID [Secret]

// Main NPC :: mal_jerun
//============================================================
malangdo,221,174,6	script	홀라당#mal_cash	559,{
	disable_items;
	mes "[홀라당]";
	mes "나는 야옹~대장장이 홀라당~";
	mes "제련의 달인, 홀라당~";
	mes "저는 모로코에서 배운 특별한 고양이 홀라당입니다~";
	mes "내 딸은 내가 너무 자랑스러워요, 홀라당~";
	mes "홀라당~ 오늘은 무엇을 개선해야 할까요?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1)
		set .@menu$, .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) +"-[Empty]" ) +":";
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) {
		mes "[홀라당]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "아라감 선생님이 야옹~이라고 하셨어요~";
			mes "어리석음에는 치료법이 없어...";
			break;
		case EQI_ARMOR:
			mes "여기엔 볼 게 하나도 없어, 야옹!!";
			break;
		case EQI_HAND_L:
			mes "야옹? 이 왼손으로 뭘 하길 원해요...?";
			break;
		case EQI_HAND_R:
			mes "야옹? 이 오른손으로 뭘 하길 바라는데...?";
			break;
		case EQI_GARMENT:
			mes "야옹? 아무것도 입지 않았잖아.";
			break;
		case EQI_SHOES:
			mes "캑~! 내 예민한 후각 건들지 마오야옹~.";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "야옹? 액세서리는 어디에 있지?";
			break;
		case EQI_HEAD_MID:
		case EQI_HEAD_LOW:
			mes "야옹? 머리 부분을 말하는 거야, 야옹?~";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[홀라당]";
		mes "아라감도 그런 건 다듬을 수 없잖아, 야옹.";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[홀라당]";
		mes "야옹~완벽한 정제. 아라감이 그랬어, 야옹?~";
		close;
	}
	mes "[홀라당]";
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
// Holink has different hardcoded prices
//	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );
		.@type$ = "armor";
		mes "갑옷을 선택했구나, 야옹~";

		switch(.@equip_lv) {
			case 1:
				.@price = 15000;
				break;
			default:
				// TODO:
				close;
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );
		.@type$ = "weapon";

		switch( .@equip_lv ){
			case 1: // Level 1 Weapon
				.@price = 500;
				mes "레벨 1 무기...?";
				break;
			case 2: // Level 2 Weapon
				.@price = 2000;
				mes "야옹, 2레벨 무기라니...?";
				break;
			case 3: // Level 3 Weapon
				.@price = 20000;
				mes "야옹야옹~~ 3레벨 무기~~";
				break;
			case 4: // Level 4 Weapon
				.@price = 50000;
				mes "야옹!...4레벨 무기라니...!";
				mes "그 동안 딱 두 번 봤어요";
				mes "아라감에게서 배운다... 나야옹!!";
				break;
			default:
				// TODO:
				close;
		}
	}else{
		// TODO:
		close;
	}

	mes "^ff9999"+ getitemname(.@material) +"^000000 및 ^ff9999"+ .@price +"^000000 이 개선에는 제니가 필요해요, 야옹~";
	mes "계속할래, 야옹?~";
	next;
	if(select("예!!:아니요!!") == 2) {
		mes "[홀라당]";
		mes "캑!!";
		mes "제련 마스터 호링크를 믿지 않는구나, 야옹~";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[홀라당]";
		mes "Meow!!";
		if (.@type$ == "armor")
			mes "이 방어구는 이미 몇 번이나 다듬어졌구나, 야옹.";
		else {
			mes "위험. 위험~";
			mes "이 무기는 많이 다듬어졌어, 야옹~";
			next;
			mes "[홀라당]";
		}
		mes "계속하면 깨질 수도 있어요";
		mes "이 아이템을 좀 더 다듬어 보려고요, 야옹.";
		next;
		mes "[홀라당]";
		mes ""+ .@type$ +"이 깨지면, 할 수 있습니다";
		mes "다시는 사용하지 마세요, 야옹. 말할 것도 없이... 모두 현재";
		mes "^ff0000카드와 마법은 반드시 사라집니다^000000.";
		mes "아직도 해보고 싶냐, 야옹~?";
		next;
		if(select("네, 그렇습니다!!:잊으세요!!") == 2) {
			mes "[홀라당]";
			mes "야옹! 현명한 선택이야, 야옹.";
			mes "하지만!!";
			mes "네가 제련의 달인 호링크를 의심하는 모습이 안 좋아, 야옹~";
			close;
		}
	}
	if (countitem(.@material) == 0 || Zeny < .@price) {
		mes "[홀라당]";
		mes "재료가 없습니다.";
		mes "^ff9999가 필요합니다"+ getitemname(.@material) +"^000000 및 ^ff9999"+ .@price +"^000000 제니, 야옹~";
		mes "가서 가져와야옹~";
		close;
	}
	delitem .@material,1;
	set Zeny, Zeny-.@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[홀라당]";
		emotion ET_FRET;
		mes "잠깐만요...";
		mes "내가 바보라고 생각해요?!";
		mes "내가 보지 않는 동안 아이템을 바꾸셨네요! 여기서 나가세요!";
		close;
	}

	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		mes "[홀라당]";
		mes "나~나~야옹! 재미있고 즐겁게 다듬어~";
		next;
		emotion ET_CHUP;
		mes "[홀라당]";
		mes "완벽해!! 완벽해, 야옹!!";
		mes "저는 제련마법사 아라감 님의 제자입니다~";
		mes "홀라당!!";
		mes "또 성공적인 정제의 날이야, 야옹!!";
		close;
	}
	failedrefitem .@part;
	mes "[홀라당]";
	mes "메오~야옹~캬아악!!";
	next;
	switch(rand(1,5)) {
		case 1: emotion ET_CRY; break;
		case 2: emotion ET_PROFUSELY_SWEAT; break;
		case 3: emotion ET_KEK; break;
		case 4: emotion ET_SCRATCH; break;
		case 5: emotion ET_BIGTHROB; break;
	}
	mes "[홀라당]";
	mes "야옹!! 아아아악~~!!!!";
	mes "캬아악!! 실패했어 야옹!!";
	next;
	mes "[홀라당]";
	mes "......";
	mes "......";
	mes "다~ 다~ 깨져, 야옹...";
	next;
	mes "[홀라당]";
	mes "야옹.... 아라감 선생님께서 한번 말씀하셨죠,";
	mes "실패로부터 교훈을 얻으세요...";
	mes "인간아, 이번 실패가 미래의 성공의 시작이 될 것이다.";
	close;
}
