moc_para01,22,16,5	script	카탈로그 자판기#catal02	2_VENDING_MACHINE1,{
	if (!checkweight(1301,3)) {
		mes "- 항목이 너무 많습니다. 공간을 만들어주세요. -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[카탈로그 자판기]";
	mes "구형 만능 카탈로그 실버를 1회 1장 구매할실 수 있습니다. 대량구매나 신형 특수 카탈로그 실버를 구매하실 고객님께서는 알베르타  상인조합으로 찾아와 주시기 바랍니다.";
	next;
	switch(select("만능 카탈로그 실버를 산다:그만둔다")) {
		case 1:
			mes "[카탈로그 자판기]";
			mes "200z 투입해 주십시오.";
			next;
			switch(select("투입:그만둔다")) {
				case 1:
					if (Zeny < 200) {
						mes "돈이 부족합니다.";
						close;
					} else {
						mes "구매 감사드립니다.";
						set Zeny, Zeny - 200;
						getitem 12580,1;
						end;
					}
				case 2:
					mes "[카탈로그 자판기]";
					mes "다음에 또 이용해 주십시오.";
					close;
				
				}
		case 2:
			mes "[카탈로그 자판기]";
			mes "다음에 또 이용해 주십시오.";
			close;
		}
}

			mes "장비를 매각하시면";

			close;
		}
		mes .@sell + " z를 주시면 됩니다.";
		mes "여기 있습니다.";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[카탈로그 마도사]";
		mes "저기... 50장 전부 드릴까요?";
		mes "얼나마... 필요하세요?";
		next;
		while (1) {
			input .@input;
			mes "[카탈로그 마도사]";
			if (.@input == 0) {
				mes "거래가 중지돼 버렸어요!";
				mes "이 다음엔... 어...";
				mes "다, 다시 거래 주세요!";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "50개 이하로 말씀해주세요.";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost1 * .@input;
		mes "구입하시려는 카탈로그는 총 "+ .@input + "매로";
		if (Zeny < .@sell) {
