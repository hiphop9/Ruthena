job3_guil01,91,93,3	script	희귀 독초 채집가	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- 잠깐만 !! -";
		mes "- 지금 너는 너무 많은 아이템을 -";
		mes "- 들고다니고 있어. -";
		mes "- 무게랑 소지아이템 개수를 -";
		mes "- 충분히 확인한 후 와줘. -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[희귀 독초 채집가]";
		mes "나는 세상을 떠돌며 희귀한 독초를 수집하는 사람이라네. 최근엔, 이지도르 라고 불리는 식물을 다루기 시작했지. 관심있나?";
		next;
		.@i = select("이지도르를 살려면 어떻게 해야하죠?:가축의 피와 교환한다.:쓰디쓴 풀로 교환한다.:맹독초와 교환한다.:얼어붙은 장미로 교환한다:멘트로 교환한다.:히나레로 교환한다.")-2;
		if (.@i == -1) {
			mes "[희귀 독초 채집가]";
			mes "단 돈 10,000제니와 가축의 피, 쓰디쓴 풀, 맹독초, 얼어붙은 장미, 멘트, 히나레 중 하나를 가져오면 이지도르 한 송이와 교환해주겠네.";
			next;
			mes "[희귀 독초 채집가]";
			mes "왜 돈을 요구하냐고? 나같은 희귀 수집가를 후원한다고 생각하게. 하 하 하";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,708,703; //안티페인먼트 -> 멘트 변경
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[희귀 독초 채집가]";
			mes "얼마나 원하는가?";
			mes "살 때 숫자는 최대 숫자 2,000을 넘겨선 안된다네.";
			mes "그리고 0 을 입력하면 거래가 취소될껄세.";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[희귀 독초 채집가]";
				mes "거래가 취소되었다네.";
				close;
			}
			if (.@amount > 2000) {
				mes "[희귀 독초 채집가]";
				mes "구매개수는 2,000 보다 작아야 한다네!";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[희귀 독초 채집가]";
				mes "자네가 갖고있는 아이템의 개수가 맞는지 확인해봤나?";
				close;
			}
			mes "[희귀 독초 채집가]";
			mes "좋군. 여기 이지도르 받게나.";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[희귀 독초 채집가]";
	mes "나는 세상을 떠돌며 희귀한 독초를 수집하고 있다네. 하지만 내 느낌엔 자네같은 독과 관련 없는사람에겐 팔면 안될 것 같구만... 하 하 하";
	close;
}
