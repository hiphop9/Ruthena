-	script	#GemstoneBagger	HIDDEN_NPC,{
	mes "[포장 직원]";
	mes "가방 안에 굴러다니는 자잘한 것들을";
	mes "찾기 쉽게 압축 포장해 드립니다.";
	mes "어떤 것을 정리해 드릴까요?";
	next;
	switch(select("젬스톤 주머니 교환:그만둔다")) {
	case 1:
		mes "[포장 직원]";
		mes "젬스톤 전용 주머니를 준비하겠습니다.";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[포장 직원]";
		mes "도움이 필요하시면 또 오세요.";
		close;
	}
}
