		next;

			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[공항 직원]";
			mes "1200제니를 지불해야만 들어가실 수";
			mes "있습니다. 돈이 모자라신 것 같네요.";
			close;
		}
	}
	mes "[공항 직원]";

	mes "감사합니다.";
	close;
