y_airport,143,43,5	script	공항 직원#y_air1a::y_airport1	90,{
	mes "[공항 직원]";
	mes "안녕하세요. 저희 항공은 항상";
	mes "고객님의 편안한 여행을 위해 최선을";
	mes "다하고 있습니다.";
	mes "무엇을 도와드릴까요?";
	next;
	if (select("공항 내부로 들어간다:취소") == 1) {
		mes "[공항 직원]";
		mes "비공정의 이용 요금은 1200제니입니다.";
		mes "비공정 무료이용권을 갖고 계시면";
		mes "무료로 들어가실 수 있습니다. 공항";
		mes "내부로 들어 가시겠습니까?";
		next;
		if (select("예:아니오") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[공항 직원]";
			mes "1200제니를 지불해야만 들어가실 수";
			mes "있습니다. 돈이 모자라신 것 같네요.";
			close;
		}
	}
	mes "[공항 직원]";
	mes "다음에 또 이용해주세요.";
	mes "감사합니다.";
	close;
}
